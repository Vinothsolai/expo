// 📦 BroilBro: AI Cooking Assistant App (React Native + Expo)
// Features: Ingredient Input ➜ AI Recipe ➜ Voice Instruction

import React, { useState } from 'react';
import { View, Text, TextInput, Button, ScrollView, StyleSheet } from 'react-native';
import * as Speech from 'expo-speech';

export default function App() {
  const [ingredients, setIngredients] = useState('');
  const [recipe, setRecipe] = useState('');
  const [loading, setLoading] = useState(false);

  const getRecipe = async () => {
    setLoading(true);
    setRecipe('');

    try {
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer YOUR_OPENAI_API_KEY', // Replace with your key
        },
        body: JSON.stringify({
          model: 'gpt-3.5-turbo',
          messages: [
            {
              role: 'user',
              content: `I have ${ingredients}. Suggest a simple recipe with steps.`,
            },
          ],
          temperature: 0.7,
        }),
      });

      const data = await response.json();
      const recipeText = data.choices[0].message.content;
      setRecipe(recipeText);
    } catch (error) {
      setRecipe('Something went wrong. Try again.');
    } finally {
      setLoading(false);
    }
  };

  const speakRecipe = () => {
    if (recipe) {
      Speech.speak(recipe);
    }
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>🍳 Welcome to BroilBro</Text>
      <Text style={styles.subtitle}>Enter your ingredients below</Text>

      <TextInput
        style={styles.input}
        placeholder="e.g. 2 eggs, bread"
        value={ingredients}
        onChangeText={setIngredients}
      />

      <Button title="Get Recipe" onPress={getRecipe} disabled={loading || !ingredients} />

      {loading && <Text style={styles.loading}>Cooking up ideas...</Text>}

      {recipe && (
        <View style={styles.resultContainer}>
          <Text style={styles.resultTitle}>Your Recipe:</Text>
          <Text style={styles.recipe}>{recipe}</Text>
          <Button title="🔊 Speak It Out" onPress={speakRecipe} />
        </View>
      )}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: {
    padding: 20,
    marginTop: 50,
    backgroundColor: '#fff',
  },
  title: {
    fontSize: 28,
    fontWeight: 'bold',
    marginBottom: 10,
    textAlign: 'center',
  },
  subtitle: {
    fontSize: 18,
    marginBottom: 20,
    textAlign: 'center',
  },
  input: {
    borderColor: '#ccc',
    borderWidth: 1,
    padding: 12,
    marginBottom: 10,
    borderRadius: 10,
  },
  loading: {
    textAlign: 'center',
    marginVertical: 20,
    fontStyle: 'italic',
  },
  resultContainer: {
    marginTop: 20,
    padding: 15,
    backgroundColor: '#f9f9f9',
    borderRadius: 10,
  },
  resultTitle: {
    fontSize: 20,
    fontWeight: 'bold',
    marginBottom: 10,
  },
  recipe: {
    fontSize: 16,
    marginBottom: 15,
  },
});
